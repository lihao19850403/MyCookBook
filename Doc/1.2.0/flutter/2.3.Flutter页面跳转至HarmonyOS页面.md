# Flutter页面跳转至HarmonyOS页面

从Flutter页面跳转至HarmonyOS工程页面，需要用到MethodChannel来进行跳转，具体操作步骤为：

1.  编写HarmonyOS平台特定跳转方法(对`FlutterRouterPlugin`类进行修改)：

这里主要做了三件事：

- 定义一个HarmonyOS端的路由处理handler(`routerPushHandler`)，当有从Flutter页面跳转至HarmonyOS页面的动作时，交由此handler处理。
- 定义一个routerPushHandler的设置方法(`setRouterPushHandler()`)，HarmonyOS平台可以通过此方法手动设置一个routerPushHandler，以便跳转至不同的HarmonyOS原生页面。
- 在`onMethodCall()`方法中定义了一个`go()`方法的处理过程，以后Flutter部分跳转页面都使用这个go方法进行。

```text
import {
  FlutterPlugin,
  FlutterPluginBinding,
  MethodCall,
  MethodCallHandler,
  MethodChannel,
  MethodResult,
} from '@ohos/flutter_ohos';

/** Flutter路由跳转类型。 */
export type FlutterRouterPushHandler = (path: string, params: Record<string, Object> | undefined, result: MethodResult) => boolean;

/** FlutterRouterPlugin **/
export default class FlutterRouterPlugin implements FlutterPlugin, MethodCallHandler {
  private channel: MethodChannel | null = null;

  /** Flutter路由跳转工具，默认不处理路由，返回false。主工程可自行修改为有意义的工具。 */
  static routerPushHandler: FlutterRouterPushHandler = (path, params, result) => {
    console.log(`FlutterRouterPushHandler 默认实现。路径：${path}，参数：${JSON.stringify(params)}`);
    return false;
  };

  constructor() {
  }

  /**
   * 设置Flutter路由跳转工具。
   * @param newHandler 新的路由跳转工具。
   */
  static setRouterPushHandler(newHandler: FlutterRouterPushHandler) {
    FlutterRouterPlugin.routerPushHandler = newHandler;
  }

  getUniqueClassName(): string {
    return "FlutterRouterPlugin"
  }

  onAttachedToEngine(binding: FlutterPluginBinding): void {
    this.channel = new MethodChannel(binding.getBinaryMessenger(), "flutter_router");
    this.channel.setMethodCallHandler(this)
  }

  onDetachedFromEngine(binding: FlutterPluginBinding): void {
    if (this.channel != null) {
      this.channel.setMethodCallHandler(null)
    }
  }

  onMethodCall(call: MethodCall, result: MethodResult): void {
    if (call.method == "getPlatformVersion") {
      result.success("OpenHarmony ^ ^ ")
    } else if (call.method === 'go') {
      let path: string = call.argument('path');
      let params: Record<string, Object> | undefined = call.argument('params');
      FlutterRouterPlugin.routerPushHandler(path, params, result);
    } else {
      result.notImplemented()
    }
  }
}
```

2.  在Flutter插件工程中编写go()方法的实现：

- 首先在FlutterRouterPlatform中对go()方法进行了定义。dart代码是：

```text
import 'package:plugin_platform_interface/plugin_platform_interface.dart';

import 'flutter_router_method_channel.dart';

abstract class FlutterRouterPlatform extends PlatformInterface {
  ...
  Future<T> go<T extends Object?>(String path, {dynamic params}) {
    throw UnimplementedError('go() has not been implemented.');
  }
}
```

- 然后在MethodChannelFlutterRouter中对go()方法进行了实现。dart代码是：

```text
import 'package:flutter/foundation.dart';
import 'package:flutter/services.dart';

import 'flutter_router_platform_interface.dart';

/// An implementation of [FlutterRouterPlatform] that uses method channels.
class MethodChannelFlutterRouter extends FlutterRouterPlatform {
  ...
  @override
  Future<T> go<T extends Object?>(String path, {dynamic params}) async {
    var routerParams = {};
    routerParams['path'] = path;
    if (params != null) {
      routerParams['params'] = params;
    }
    final result = await methodChannel.invokeMethod('go', routerParams);
    return result;
  }
}

```

- 此外在RouterManager中增加了一个方法`isFlutterPage()`，用来判断要跳转的页面是否Flutter页面。dart代码是：

```text
import 'dart:convert';
import 'package:flutter/widgets.dart';

/// Widget构造方法，接收任意类型参数[params]。
typedef RouterWidgetBuilder = Widget Function(dynamic params);

/// flutter路由管理器。
/// @author li.hao85
class RouterManager {
  ...
  /// 判断给定的路由信息[routerInfo]是否指向Flutter页面。
  bool isFlutterPage(String routerInfo) {
    final String path = _getRouterPath(routerInfo);
    return _routerMap.containsKey(path);
  }
  ...
}

```

- 最后在FlutterRouter中定义了一个go()方法，供Flutter工程代码调用。这个方法会判断要跳转的目标页面类型。如果目标页面是Flutter页面，则使用Flutter的Navigator跳转；如果目标页面不在Flutter路由表中，那么就尝试跳转入原生页面。dart代码是：

```text
// You have generated a new plugin project without specifying the `--platforms`
// flag. A plugin project with no platform support was generated. To add a
// platform, run `flutter create -t plugin --platforms <platforms> .` under the
// same directory. You can also find a detailed instruction on how to add
// platforms in the `pubspec.yaml` at
// https://flutter.dev/docs/development/packages-and-plugins/developing-packages#plugin-platforms.

import 'package:flutter/material.dart';
import 'package:flutter_router/router_manager.dart';

import 'flutter_router_platform_interface.dart';

class FlutterRouter {
  ...
  /// 页面跳转方法。
  ///
  /// 根据给定的路由路径[path]和路由参数[params]跳转至目标页面。
  ///
  /// 注：如果目标页面是Flutter页面，则使用Flutter的Navigator跳转，需要借助布局上下文[context]；如果目标页面不在Flutter路由表中，那么就尝试跳转入原生页面。
  Future<T> go<T extends Object?>(BuildContext context, String path,
      {Object? params}) async {
    final bool isFlutterPage = RouterManager.instance.isFlutterPage(path);
    if (isFlutterPage) {
      Widget widget =
          RouterManager.instance.getRouterWidget(path, defaultParams: params);
      return Navigator.of(context)
          .push(MaterialPageRoute(builder: (context) => widget))
          .then((result) => result);
    } else {
      return FlutterRouterPlatform.instance
          .go<T>(path, params: params)
          .then((result) => result);
    }
  }
}

```

4.  HarmonyOS平台配置routerPushHandler：

由于引用了flutter_router.har，因此HarmonyOS工程可以调用里面的方法了。此处在EntryAbility初始化时登记一个页面跳转路由handler：

```text
...
import FlutterRouterPlugin from 'flutter_router';
...
export default class EntryAbility extends UIAbility implements ExclusiveAppComponent<UIAbility> {
  onCreate(want: Want, launchParam: AbilityConstant.LaunchParam): void {
    ...
    // 声明平台特定的Flutter页面跳转路由handler。
    FlutterRouterPlugin.setRouterPushHandler(flutterRouterHandler);
  }
}
```

其中flutterRouterHandler定义如下：

```text
import { MethodResult } from '@ohos/flutter_ohos';
import { LHWebBrowser, URL_PARAMS } from '../../pages/common/webbrowser/LHWebBrowser';

/**
 * 鸿蒙平台下的Flutter页面跳转handler。
 * @param path 路由路径。
 * @param params 路由参数。
 * @param result 跳转结果，用于向Flutter页面返回参数。
 * @returns 处理是否完成。
 */
export const flutterRouterHandler: (path: string, params: Record<string, Object> | undefined, result: MethodResult) => boolean =
  (path, params, result) => {
    if (path === 'LHWebBrowser') {
      let routerParams: Record<string, string | undefined> = {};
      if (params !== undefined) {
        let urlParams: string = params instanceof Map ? params.get(URL_PARAMS) as string : params[URL_PARAMS] as string;
        routerParams[URL_PARAMS] = urlParams;
      }
      LHWebBrowser.go(routerParams);
    }
    return true;
  };

```

这个handler先对Web页的跳转进行了处理。
